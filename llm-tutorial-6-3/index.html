策略网络和价值函数网络的定义 Python

class PolicyNet(nn.Module):
    def __init__(self, action_size):
        super().__init__()
        self.l1 = nn.Linear(4, 128)
        self.l2 = nn.Linear(128, action_size)

    def forward(self, x):
        x = F.relu(self.l1(x))
        x = F.softmax(self.l2(x), dim=1)
        return x

class ValueNet(nn.Module):
    def __init__(self):
        super().__init__()
        self.l1 = nn.Linear(4, 128)
        self.l2 = nn.Linear(128, 1)

    def forward(self, x):
        x = F.relu(self.l1(x))
        x = self.l2(x)
        return x

6.3.2 PPO算法的实现

我们来实现Agent，代码如下：

class Agent:
    def __init__(self):
        self.gamma = 0.98
        self.lr_pi = 0.001
        self.lr_v = 0.02
        self.action_size = 2

        self.pi = PolicyNet(self.action_size)
        self.v = ValueNet()

        self.optimizer_pi = optim.Adam(self.pi.parameters(), lr=self.lr_pi)
        self.optimizer_v = optim.Adam(self.v.parameters(), lr=self.lr_v)

    def get_action(self, state):
        probs = self.pi(torch.tensor(state).unsqueeze(0)).squeeze(0)
        m = Categorical(probs)
        action = m.sample().item()
        return action, probs

    def rollout(self, env):
        """采集一条轨迹"""
        state = env.reset()
        states, next_states, actions, action_probs, rewards, dones = [], [], [], [], [], []
        done = False

        while not done:
            action, probs = self.get_action(state)
            next_state, reward, done, _ = env.step(action)
            states.append(state) # s_t
            actions.append(action) # a_t
            action_probs.append(probs[action]) # π(a_t|s_t)
            next_states.append(next_state) # s_{t+1}
            rewards.append(reward) # R_t
            dones.append(done) # done_t
            state = next_state # 状态转移

        return states, next_states, actions, action_probs, rewards, dones

接下来我们要实现智能体的更新（update）代码，在PPO算法中，我们需要用到旧策略采样的轨迹。所以需要保存旧策略采样的轨迹的完整信息。

我们采样的轨迹是：

$$\tau = [s_0, a_0, R_0, s_1, a_1, R_1, \cdots, s_T, a_T, R_T] \quad (6.12)$$

那么我们使用元组来保存一条轨迹的信息。结构如下

$$\text{一条轨迹的详细信息} = \left\{ \begin{array}{l} \text{states} : [s_0, s_1, ..., s_{T-1}] \\ \text{next\_states} : [s_1, s_2, ..., s_T] \\ \text{actions} : [a_0, a_1, ..., a_{T-1}] \\ \text{action\_probs} : [\pi_{\theta_{old}}(a_0|s_0), \pi_{\theta_{old}}(a_1|s_1), ..., \pi_{\theta_{old}}(a_{T-1}|s_{T-1})] \\ \text{rewards} : [R_0, R_1, ..., R_{T-1}] \\ \text{dones} : [\text{False, False, ..., True}] \end{array} \right. \quad (6.13)$$

有了旧策略采样的轨迹的所有详细信息，我们就可以来实现PPO算法了。代码如下。

class Agent:
    ...

    def update(self, trajectory):
        states, next_states, actions, action_probs, rewards, dones = trajectory
        # [s_0, s_1, ..., s_{T-1}]
        states = torch.tensor(states)
        # [a_0, a_1, ..., a_{T-1}]
        actions = torch.tensor(actions).view(-1, 1)
        # [R_0, R_1, ..., R_{T-1}]
        rewards = torch.tensor(rewards).view(-1, 1)
        # [s_1, s_2, ..., s_T]
        next_states = torch.tensor(next_states)
        # [False, False, ..., True]
        dones = torch.tensor(dones, dtype=torch.float).view(-1, 1)

        # [V(s_0), V(s_1), ..., V(s_{T-1})]
        V = self.v(states).detach()
        # TD-target = R_t + γV(s_{t+1})
        # [TD-target_0, TD-target_1, ..., TD-target_{T-1}]
        td_target = rewards + self.gamma * self.v(next_states) * (1 - dones)
        # 一步TD误差：δ_t = R_t + γV(s_{t+1}) - V(s_t)
        # [δ_0, δ_1, ..., δ_{T-1}]
        td_delta = td_target - V
        # 计算每个时刻t的广义优势估计（GAE），gae是常数，不在计算图中
        # [A^GAE_0, A^GAE_1, ..., A^GAE_{T-1}]
        gae = compute_gae(self.gamma, td_delta.cpu())
        # 冻结一份旧策略采取动作的对数概率logπ_old(a|s)
        # [logπ_old(a_0|s_0), logπ_old(a_1|s_1), ..., logπ_old(a_{T-1}|s_{T-1})]
        old_probs = torch.tensor(action_probs).view(-1, 1)
        old_log_probs = torch.log(old_probs).detach()
        # 每条轨迹使用10次，也就是更新10次策略网络
        for _ in range(10):
            # 新策略采取动作的对数概率：logπ_θ(a|s)
            # [logπ_θ(a_0|s_0), logπ_θ(a_1|s_1), ..., logπ_θ(a_{T-1}|s_{T-1})]
            log_probs = torch.log(self.pi(states).gather(1, actions))
            # 计算比值：ρ(θ) = π_θ(a|s) / π_old(a|s)
            ratio = torch.exp(log_probs - old_log_probs)
            # ρ(θ)A^GAE
            # [ρ_0 A^GAE_0, ρ_1 A^GAE_1, ..., ρ_{T-1} A^GAE_{T-1}]
            surr1 = ratio * gae
            # clip(ρ(θ), 1-ε, 1+ε)A^GAE
            # [clip(ρ_0)A^GAE_0, clip(ρ_1)A^GAE_1, ..., clip(ρ_{T-1})A^GAE_{T-1}]
            surr2 = torch.clamp(ratio, 1 - 0.2, 1 + 0.2) * gae
            # π loss = -1/T Σ min(surr1, surr2)
            pi_loss = torch.mean(-torch.min(surr1, surr2))
            # V loss = 1/T Σ (V(s_t) - (A^GAE_t + V(s_t)))²
            # 这里的'gae+V'是常数，不在计算图中
            v_loss = torch.mean(F.mse_loss(self.v(states), gae + V))
            self.optimizer_pi.zero_grad()
            self.optimizer_v.zero_grad()
            pi_loss.backward()
            v_loss.backward()
            self.optimizer_pi.step()
            self.optimizer_v.step()

比值的计算方法

$$\because A = e^{\log A}$$ $$\therefore \frac{A}{B} = e^{\log A - \log B} = \frac{e^{\log A}}{e^{\log B}}$$

价值网络的损失 MSE

如果使用单步TD误差，那么

$$V_{\omega}(s_t) \stackrel{\text{def}}{=} \text{TD Target} = \text{TD Error} + V_{\omega_{\text{old}}}(s_t) = R_t + \gamma V_{\omega_{\text{old}}}(s_{t+1}) \quad (6.15)$$

如果使用广义优势估计，那么

$$V_{\omega}(s_t) \stackrel{\text{def}}{=} \text{GAE Target} = \text{GAE} + V_{\omega_{\text{old}}}(s_t) \quad (6.16)$$

TD Target 和 GAE Target 要从计算图中剔除，作为常数使用。

6.3.3 广义优势估计的计算

def compute_gae(gamma, td_delta):
    # δ_t
    td_delta = td_delta.detach().numpy()
    gae_list = []
    last_gae = 0.0
    lam = 0.95
    # A_t = δ_t + γλA_{t+1}
    for delta in td_delta[::-1]:
        last_gae = gamma * lam * last_gae + delta
        gae_list.append(last_gae)
    gae_list.reverse()
    return torch.tensor(gae_list)

6.3.4 训练循环

def train(env, agent):
    return_list = []
    episode_list = []
    for episode in range(500):
        trajectory = agent.rollout(env)
        agent.update(trajectory)
        # 统计信息
        episode_reward = sum(trajectory[4])
        return_list.append(episode_reward)
        episode_list.append(episode)

        if (episode + 1) % 10 == 0:
            print(f"回合: {episode}, 回报: {episode_reward}")

    return return_list, episode_list

def main():
    env = gym.make("CartPole-v0")
    env.seed(0)
    torch.manual_seed(0)
    agent = Agent()
    return_list, episode_list = train(env, agent)
    plot_loss(episode_list, return_list, "ppo-loss.pdf")
