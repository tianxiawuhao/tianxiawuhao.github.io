使用java 实现mqtt两种方式

Posted on 2023-08-21，9 min read

开发 MQTT 客户端通常有两种方式：一种是使用 Paho Java 原生库，另一种是使用 Spring Boot 集成。

选哪种方式之前，先把发布/订阅模型和 QoS 语义弄清楚会更省事：两种实现底层都是 Paho，差别只在于谁帮你管理连接、订阅和消息转换。下图是本文反复用到的两个关键点：消息如何经由 Broker 流转，以及 QoS 0/1/2 各自的确认流程。

Paho Java 库实现

Eclipse Paho Java Client 是用 Java 编写的 MQTT 客户端库，可用于 JVM 或其他 Java 兼容平台（例如 Android）。它提供两套 API，先分清两者的区别：

  • MqttClient（同步）：connect、subscribe、publish 会阻塞到动作完成（QoS 1/2 时还要等到确认），代码直观，但每一次收发都占用调用线程，适合简单脚本、边缘端小工具或低频收发。
  • MqttAsyncClient（异步）：所有动作立即返回一个 IMqttToken，可以通过 IMqttActionListener 回调或 token.waitForCompletion() 等待结果，不会阻塞业务线程，是生产环境常用的形态。

通过 Maven 引入 Paho Java：

<dependency>
  <groupId>org.eclipse.paho</groupId>
  <artifactId>org.eclipse.paho.client.mqttv3</artifactId>
  <version>1.2.2</version>
</dependency>

Paho Java 使用示例

Java 体系中 Paho Java 是比较稳定、广泛应用的 MQTT 客户端库。下面的示例使用同步 API 连接公共 Broker（broker.emqx.io）并完成订阅与发布：

package io.emqx;

import org.eclipse.paho.client.mqttv3.MqttClient;
import org.eclipse.paho.client.mqttv3.MqttConnectOptions;
import org.eclipse.paho.client.mqttv3.MqttException;
import org.eclipse.paho.client.mqttv3.MqttMessage;
import org.eclipse.paho.client.mqttv3.persist.MemoryPersistence;


public class App {
    public static void main(String[] args) {
        String subTopic = "testtopic/#";
        String pubTopic = "testtopic/1";
        String content = "Hello World";
        int qos = 2;
        String broker = "tcp://broker.emqx.io:1883";
        // clientId 在同一个 Broker 上必须唯一，重复会互相踢线
        String clientId = "emqx_test_" + System.currentTimeMillis();
        MemoryPersistence persistence = new MemoryPersistence();

        try {
            MqttClient client = new MqttClient(broker, clientId, persistence);

            // MQTT 连接选项
            MqttConnectOptions connOpts = new MqttConnectOptions();
            connOpts.setUserName("emqx_test");
            connOpts.setPassword("emqx_test_password".toCharArray());
            // cleanSession=true：不保留会话，重连后需要重新订阅
            connOpts.setCleanSession(true);
            connOpts.setKeepAliveInterval(60);
            connOpts.setAutomaticReconnect(true);

            // 设置回调
            client.setCallback(new OnMessageCallback());

            // 建立连接（同步 API 会阻塞到连接完成）
            System.out.println("Connecting to broker: " + broker);
            client.connect(connOpts);

            System.out.println("Connected");
            System.out.println("Publishing message: " + content);

            // 订阅：第二个参数是订阅 QoS，这里与发布 QoS 保持一致
            client.subscribe(subTopic, qos);

            // 消息发布所需参数
            MqttMessage message = new MqttMessage(content.getBytes());
            message.setQos(qos);
            client.publish(pubTopic, message);
            System.out.println("Message published");

            // 主动断开不会触发遗嘱消息
            client.disconnect();
            System.out.println("Disconnected");
            client.close();
            // 演示程序主动退出，否则 Paho 的心跳线程会让 JVM 一直存活；正式服务不要这样写
            System.exit(0);
        } catch (MqttException me) {
            System.out.println("reason " + me.getReasonCode());
            System.out.println("msg " + me.getMessage());
            System.out.println("loc " + me.getLocalizedMessage());
            System.out.println("cause " + me.getCause());
            System.out.println("excep " + me);
            me.printStackTrace();
        }
    }

}

两个容易踩的点：clientId 用了时间戳后缀，避免多实例互相顶掉；client.subscribe(topic, qos) 显式写出订阅 QoS，比单参数的 subscribe(topic)（默认 QoS 1）更不容易产生误会。

回调消息处理类 OnMessageCallback.java

package io.emqx;

import org.eclipse.paho.client.mqttv3.IMqttDeliveryToken;
import org.eclipse.paho.client.mqttv3.MqttCallbackExtended;
import org.eclipse.paho.client.mqttv3.MqttException;
import org.eclipse.paho.client.mqttv3.MqttMessage;

/**
 * 实现 MqttCallbackExtended 而不是 MqttCallback：
 * 开启 automaticReconnect 后，重连成功不会自动恢复订阅，需要在 connectComplete 里重新订阅。
 */
public class OnMessageCallback implements MqttCallbackExtended {

    public void connectComplete(boolean reconnect, String serverURI) {
        System.out.println(reconnect ? "重连成功，需要重新订阅" : "首次连接成功");
        // 此处应调用 client.subscribe(topic, qos) 恢复订阅
    }

    public void connectionLost(Throwable cause) {
        // 连接丢失后触发；开启 automaticReconnect 时交给客户端重连，这里只记录日志
        System.out.println("连接断开，可以做重连" + cause);
    }

    public void messageArrived(String topic, MqttMessage message) throws Exception {
        // subscribe 后得到的消息会执行到这里面，运行在 Paho 的回调线程上，不要做耗时操作
        System.out.println("接收消息主题:" + topic);
        System.out.println("接收消息Qos:" + message.getQos());
        System.out.println("接收消息内容:" + new String(message.getPayload()));
    }

    public void deliveryComplete(IMqttDeliveryToken token) {
        System.out.println("deliveryComplete---------" + token.isComplete());
    }

}

至此，就实现了简单的 MQTT 连接与消息收发。需要注意 messageArrived 是在客户端内部线程上串行执行的：在里面做数据库写入、HTTP 调用这类耗时操作，会拖慢 PUBACK 的发送和心跳，进而被 Broker 判定超时断线。稳妥的做法是只做解析和投递，把业务处理交给自己的线程池，并保证处理逻辑幂等（QoS 1 可能重复投递）。

MqttAsyncClient 异步实现

同步 API 在高频收发场景下会拖住业务线程，改用 MqttAsyncClient 更合适。它把「发起动作」和「等待结果」拆开，并且可以在 connectComplete 里统一恢复订阅：

package io.emqx;

import org.eclipse.paho.client.mqttv3.IMqttActionListener;
import org.eclipse.paho.client.mqttv3.IMqttDeliveryToken;
import org.eclipse.paho.client.mqttv3.IMqttToken;
import org.eclipse.paho.client.mqttv3.MqttAsyncClient;
import org.eclipse.paho.client.mqttv3.MqttCallbackExtended;
import org.eclipse.paho.client.mqttv3.MqttConnectOptions;
import org.eclipse.paho.client.mqttv3.MqttException;
import org.eclipse.paho.client.mqttv3.MqttMessage;
import org.eclipse.paho.client.mqttv3.persist.MemoryPersistence;

public class AsyncApp {

    private static final String SUB_TOPIC = "testtopic/#";

    public static void main(String[] args) throws MqttException {
        String broker = "tcp://broker.emqx.io:1883";
        final MqttAsyncClient client =
                new MqttAsyncClient(broker, "emqx_async_" + System.currentTimeMillis(), new MemoryPersistence());

        MqttConnectOptions options = new MqttConnectOptions();
        options.setUserName("emqx_test");
        options.setPassword("emqx_test_password".toCharArray());
        // false：持久会话，Broker 会为离线期间的 QoS 1/2 消息排队（clientId 必须固定，否则会话拿不回来）
        options.setCleanSession(false);
        options.setKeepAliveInterval(60);
        options.setConnectionTimeout(10);
        // 断线后自动重连：1s 起指数退避，上限 128s
        options.setAutomaticReconnect(true);
        // 遗嘱消息：客户端异常掉线时由 Broker 代发，必须在 connect 之前设置
        options.setWill("testtopic/status", "offline".getBytes(), 1, true);

        client.setCallback(new MqttCallbackExtended() {
            public void connectComplete(boolean reconnect, String serverURI) {
                try {
                    // 重连成功后必须重新订阅，Broker 不会替你恢复
                    client.subscribe(SUB_TOPIC, 1);
                } catch (MqttException e) {
                    e.printStackTrace();
                }
            }

            public void connectionLost(Throwable cause) {
                System.out.println("连接断开：" + cause);
            }

            public void messageArrived(String topic, MqttMessage message) {
                System.out.println(topic + " -> " + new String(message.getPayload()));
            }

            public void deliveryComplete(IMqttDeliveryToken token) {
            }
        });

        // 异步连接：不阻塞当前线程，结果通过回调返回
        client.connect(options, null, new IMqttActionListener() {
            public void onSuccess(IMqttToken asyncActionToken) {
                try {
                    MqttMessage message = new MqttMessage("Hello World".getBytes());
                    message.setQos(1);
                    client.publish("testtopic/1", message);
                } catch (MqttException e) {
                    e.printStackTrace();
                }
            }

            public void onFailure(IMqttToken asyncActionToken, Throwable exception) {
                exception.printStackTrace();
            }
        });
    }
}

连接参数、遗嘱消息与断线重连

Paho 的连接参数不多，但每一个都直接影响可靠性，逐个说清楚：

  • cleanSession：默认 true，每次连接都新建会话，Broker 不保存订阅关系，也不为离线客户端排队消息；设为 false 表示持久会话，Broker 会保留订阅关系以及未确认的 QoS 1/2 消息。持久会话必须配合固定的 clientId，否则重连时按新客户端处理，会话等于白留；同时 Broker 侧会一直占用内存，长期离线的会话最终仍会被清理。
  • keepAliveInterval：心跳间隔，单位秒，默认 60。客户端在没有其他报文时按此间隔发送 PINGREQ，Broker 通常在 1.5 倍时间内没收到任何报文就判定掉线并发布遗嘱。设为 0 表示关闭心跳，一般不推荐。
  • connectionTimeout：连接超时，单位秒，默认 30。注意 0 是合法值，含义是禁用超时处理（一直等到连接成功或失败），并不是「必须大于 0」。
  • automaticReconnect：Paho 1.2.0 起支持，重连延迟从 1 秒开始指数退避，最大 128 秒。它只负责重建 TCP 连接与协议会话，不会自动恢复订阅，也不会重发应用层未确认的消息，所以要在 connectComplete 里重新订阅。
  • 遗嘱消息（Will）：通过 setWill(topic, payload, qos, retained) 设置，必须写在 connect 之前。只有异常断线（心跳超时、网络中断、没有走正常 DISCONNECT）时 Broker 才代为发布；主动调用 disconnect() 不会触发遗嘱。所以「在线/离线状态」这类主题适合用遗嘱消息维护。

再说几个真实项目里最容易出问题的资源与内存点：

  1. 重连时不断 new 客户端：在 connectionLost 里直接 new MqttClient(...) 而不关闭旧对象，会持续泄漏 socket、心跳线程和回调线程；如果 clientId 相同，新旧连接还会互相踢线，表现为「一连上就掉」。正确做法是复用同一个客户端实例，交给 automaticReconnect 处理，或者旧实例先 disconnect() + close() 再新建。
  2. 忘记释放：disconnect() 与 close() 要成对调用，Spring 中放在 @PreDestroy 或 DisposableBean 里；否则应用重启后旧会话与线程仍在。
  3. 僵尸会话：用 cleanSession=false 加随机 clientId 做压测，会在 Broker 上留下大量永远不会再被使用的会话，占用内存和磁盘。
  4. 动态订阅不取消：按设备、按用户动态订阅却不 unsubscribe，客户端和 Broker 的订阅表会无限增长。
  5. 在途消息过多：QoS 1/2 的未确认消息默认最多 10 条（setMaxInflight），调大能提高吞吐，但会同时放大内存占用和重复投递的概率。

QoS 语义与选择

QoS 决定「消息送达」的强度，也决定开销，三种级别的差别就是确认握手的次数：

QoS语义握手流程代价与风险
0最多一次（尽力而为）PUBLISH延迟最低；网络抖动、客户端离线即丢消息
1至少一次PUBLISH → PUBACK未收到确认会重发，订阅端可能收到重复消息
2恰好一次PUBLISH → PUBREC → PUBREL → PUBCOMP两次往返，延迟与资源开销最高

落到代码上还有几条容易忽略的规则：

  • 实际投递 QoS = min(发布消息的 QoS, 订阅时申请的 QoS)。发布端设了 QoS 2，订阅端只申请 QoS 0，最终仍按 QoS 0 投递。
  • QoS 只覆盖客户端到 Broker 这一段链路。Broker 之后的转发、下游系统的处理都可能丢消息，端到端的可靠要靠业务侧补偿，而不是把 QoS 调到 2 就完事。
  • 离线消息依赖持久会话。只有 cleanSession=false 时，Broker 才会为离线客户端保留 QoS 1/2 消息；QoS 0 消息对离线客户端直接丢弃。
  • 保留消息（retained）与 QoS 无关。Broker 为每个主题保存最后一条 retained 消息，新订阅者一订阅就能立刻收到。
  • 选择建议：高频状态数据用 QoS 0；指令、事件用 QoS 1 并在消费端做幂等；只有计费、不可重复执行的动作才值得用 QoS 2。

Spring Boot 集成 MQTT

Spring Integration 提供了入站适配器和出站适配器来支持 MQTT 协议，底层依然是 Paho。本节的示例基于 Spring Boot 2.2.2 与 spring-integration 5.4.3：入站方向用 MqttPahoMessageDrivenChannelAdapter 订阅主题并把消息送进通道，出站方向用 MqttPahoMessageHandler 发送消息。

核心代码

首先是配置类，它负责创建连接工厂并绑定上面的配置项。@ConfigurationProperties 需要 setter 才能注入，这里由 Lombok 的 @Data 提供：

@Data
@Configuration
@ConfigurationProperties(prefix = "spring.mqtt")
public class MqttConfiguration {

    private String username;
    private String password;
    private String url;
    private String clientId;
    private String topic = "TOPIC_DEFAULT";
    private Integer completionTimeout = 2000;

    /**
     * 注册 MQTT 客户端工厂
     */
    @Bean
    public MqttPahoClientFactory mqttClientFactory() {
        DefaultMqttPahoClientFactory factory = new DefaultMqttPahoClientFactory();
        MqttConnectOptions options = new MqttConnectOptions();
        // cleanSession=false 时，客户端、服务端重启后仍保持状态：订阅关系被视为持久订阅，
        // QoS 1/2 消息在离线期间也会被保留；true 则不保留任何状态。
        options.setCleanSession(false);
        // 连接超时，单位秒，默认 30；0 表示禁用超时处理（一直等到连接成功或失败）。
        options.setConnectionTimeout(30);
        // 心跳间隔，单位秒，必须大于 0；发送与接收之间的最大空闲间隔。
        options.setKeepAliveInterval(90);
        // 自动重连（重连后订阅由 adapter 的 recoveryInterval 负责恢复）
        options.setAutomaticReconnect(true);
        if (this.getUsername() != null) {
            options.setUserName(this.getUsername());
            // 只配用户名不配密码会让这里抛 NullPointerException
            options.setPassword(this.getPassword().toCharArray());
        }
        options.setServerURIs(new String[]{this.getUrl()});

        factory.setConnectionOptions(options);
        return factory;
    }

}

入站配置：适配器订阅主题，把消息投递到通道，再由 MessageHandler 处理。消息驱动的适配器会自动重连，setRecoveryInterval 决定重连间隔：

@Slf4j
@AllArgsConstructor
@Configuration
@IntegrationComponentScan
public class MqttInboundConfiguration {

    private MqttConfiguration mqttConfig;
    private MqttPahoClientFactory factory;
    private MqttMessageReceiver mqttMessageReceiver;

    /**
     * 此处可以使用其他消息通道
     * Spring Integration 默认的消息通道，允许将消息发送给一个订阅者，并阻塞发送方直到消息被接收。
     */
    @Bean
    public MessageChannel mqttInBoundChannel() {
        return new DirectChannel();
    }

    /**
     * 适配器：客户端作为消费者订阅主题并消费消息
     * clientId 加上时间戳后缀，避免同一 Broker 上多个实例互相踢线
     */
    @Bean
    public MessageProducerSupport mqttInbound() {
        MqttPahoMessageDrivenChannelAdapter adapter =
                new MqttPahoMessageDrivenChannelAdapter(
                        mqttConfig.getClientId() + "-" + System.currentTimeMillis(),
                        factory,
                        mqttConfig.getTopic());

        adapter.setCompletionTimeout(60000);
        adapter.setConverter(new DefaultPahoMessageConverter());
        // 连接异常时每 10 秒尝试恢复
        adapter.setRecoveryInterval(10000);
        // 订阅 QoS；最终投递 QoS 取它与发布 QoS 的较小值
        adapter.setQos(0);
        adapter.setOutputChannel(mqttInBoundChannel());
        return adapter;
    }

    /**
     * MQTT 入站消息处理器：从入站通道接收生产者发来的消息并处理
     */
    @Bean
    @ServiceActivator(inputChannel = "mqttInBoundChannel")
    public MessageHandler mqttMessageHandler() {
        return this.mqttMessageReceiver;
    }

}

@Slf4j
@AllArgsConstructor
@Component
public class MqttMessageReceiver implements MessageHandler {

    @Override
    public void handleMessage(Message<?> message) throws MessagingException {
        try {
            MessageHeaders headers = message.getHeaders();
            // 获取消息 Topic
            String receivedTopic = (String) headers.get(MqttHeaders.RECEIVED_TOPIC);
            // 获取该条消息实际的 QoS
            Integer receivedQos = (Integer) headers.get(MqttHeaders.RECEIVED_QOS);
            log.info("[获取到的消息 topic：]{}，qos：{}", receivedTopic, receivedQos);
            // 获取消息体
            String payload = (String) message.getPayload();
            log.info("[获取到的消息 payload：]{}", payload);
            // todo ....
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

}

出站配置与发送者：

@Slf4j
@AllArgsConstructor
@Configuration
public class MqttOutboundConfiguration {

    private MqttConfiguration mqttConfig;
    private MqttPahoClientFactory factory;

    @Bean
    public MessageChannel mqttOutboundChannel() {
        return new DirectChannel();
    }

    @Bean
    @ServiceActivator(inputChannel = "mqttOutboundChannel")
    public MessageHandler mqttOutbound() {
        MqttPahoMessageHandler messageHandler = new MqttPahoMessageHandler(
                mqttConfig.getClientId() + "-out-" + System.currentTimeMillis(), factory);

        messageHandler.setDefaultQos(0);
        // 开启异步：发送方法立即返回，失败不会抛给调用方
        messageHandler.setAsync(true);
        messageHandler.setDefaultTopic(mqttConfig.getTopic());
        return messageHandler;
    }

}

@Component
@MessagingGateway(defaultRequestChannel = "mqttOutboundChannel")
public interface MqttGateway {

    /**
     * 发送 mqtt 消息
     *
     * @param topic 主题
     * @param payload 内容
     */
    void sendToMqtt(@Header(MqttHeaders.TOPIC) String topic, String payload);

    /**
     * 发送包含 qos 的消息
     *
     * @param topic 主题
     * @param qos 消息质量等级：
     *            QoS 0 最多发送一次，不保证送达，消息可能丢失；
     *            QoS 1 至少送达一次，会重试直到收到确认，但订阅者可能收到重复消息；
     *            QoS 2 通过两次往返的去重流程，确保订阅者恰好收到一次消息。
     * @param payload 消息体
     */
    void sendToMqtt(@Header(MqttHeaders.TOPIC) String topic, @Header(MqttHeaders.QOS) int qos, String payload);

    /**
     * 发送包含 qos 的字节消息
     *
     * @param topic 主题
     * @param qos 消息质量等级，含义同上
     * @param payload 消息体
     */
    void sendToMqtt(@Header(MqttHeaders.TOPIC) String topic, @Header(MqttHeaders.QOS) int qos, byte[] payload);

}

@Component
@AllArgsConstructor
public class MqttMessageSender {

    private MqttGateway mqttGateway;

    /**
     * 发送 mqtt 消息
     *
     * @param topic 主题
     * @param message 内容
     */
    public void send(String topic, String message) {
        mqttGateway.sendToMqtt(topic, message);
    }

    /**
     * 发送包含 qos 的 JSON 消息（JSONObject 由 fastjson/jackson 提供，注意 import）
     *
     * @param topic 主题
     * @param qos 质量
     * @param messageBody 消息体
     */
    public void send(String topic, int qos, JSONObject messageBody) {
        mqttGateway.sendToMqtt(topic, qos, messageBody.toString());
    }

    /**
     * 发送包含 qos 的字节消息
     *
     * @param topic 主题
     * @param qos 质量
     * @param message 消息体
     */
    public void send(String topic, int qos, byte[] message) {
        mqttGateway.sendToMqtt(topic, qos, message);
    }

}

以上就是一个可用的收发闭环。有两个细节值得注意：出站 setAsync(true) 之后，发送失败不会以异常形式抛给调用方，必须自己接 MqttCallback 或看日志，否则消息丢了也不知道；入站侧的 QoS 建议与发布侧对齐，避免出现「发布 QoS 2、订阅 QoS 0」这种被降到 QoS 0 的情况。

两种方式对比与选型

维度原生 PahoSpring Integration MQTT
依赖只需 paho 客户端 jarspring-integration-mqtt 及集成容器
编程模型手动连接、订阅、回调Channel + Adapter + MessageHandler，声明式
断线重连依赖 automaticReconnect，订阅需自己恢复recoveryInterval 自动重建连接与订阅
消息转换自己做 byte[] 编解码可插拔 MessageConverter（String/JSON/byte[]）
可控性线程、QoS、在途消息数完全可控受通道类型影响，DirectChannel 默认同步调用
适用场景网关、边缘端、需要精细控制的场景Spring Boot 业务系统内部收发消息

选型建议：如果项目已经是 Spring Boot 且只是业务系统之间收发消息，直接用 Spring Integration，省掉连接管理和重连逻辑；如果是设备网关、需要自己控制线程模型和吞吐，用原生 Paho。无论哪种方式，clientId 唯一、连接参数统一、异常断线与遗嘱消息、消费端幂等这四件事都必须处理，它们和用哪套 API 无关。
