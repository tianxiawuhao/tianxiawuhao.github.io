工具类

「工具类」是项目里复用度最高、也最容易失控的一类代码：同一个功能，有人用 StringUtils.isEmpty，有人用 StrUtil.isBlank，有人手写 a == null || a.length() == 0；同一个判空，写法能有五六种。结果就是依赖越引越多，边界行为各不相同，出问题时排查成本很高。

下面按用途把工具类分成四类，分别给出这一篇里收录的 HttpUtil、JavaScriptUtil、MinioUtil 三个实现，以及它们各自需要留意的坑。

通用原则只有两条：能只用 JDK 和 Spring 自带能力解决的，就不要引入第三方库；确实要引，就全项目统一一套，禁止同一功能同时存在多套实现。判空和默认值的选择路径可以归纳成下面这张图。

HttpUtil

下面这个 HttpUtil 把 RestTemplate 包了一层，统一设置 JSON 请求头、统一把异常翻译成业务错误码。它要作为一个 @Component 交给 Spring 管理，同时又要提供静态调用入口，所以用 @PostConstruct 把容器里的 restTemplate 回填到静态字段上。

import cn.hutool.json.JSONObject;
import cn.hutool.json.JSONUtil;
import dt.logincenter.exception.BusinessException;
import dt.logincenter.exception.basic.ResponseCode;
import lombok.extern.slf4j.Slf4j;
import org.springframework.http.*;
import org.springframework.stereotype.Component;
import org.springframework.web.client.RestTemplate;

import javax.annotation.PostConstruct;
import javax.annotation.Resource;
import java.util.Iterator;
import java.util.Map;

@Component
@Slf4j
public class HttpUtil {

    @Resource
    private RestTemplate restTemplate;

    private static HttpUtil httpUtil;

    @PostConstruct
    public void init(){
        httpUtil = this;
        httpUtil.restTemplate = this.restTemplate;
    }

    /**
     * 发送请求Get请求
     *
     * @param url   请求接口
     * @param param 请求参数
     * @param paramType 请求参数类型 json：false  String：true
     */
    public JSONObject sendGet(String url, String param, boolean paramType) throws Exception {
        JSONObject jsonObject = null;
        try {
            //设置请求头
            HttpHeaders headers = new HttpHeaders();
            headers.setContentType(MediaType.APPLICATION_JSON);

            ResponseEntity<String> resEntity = null;
            if (!paramType) {
                HttpEntity httpEntity = new HttpEntity<>(param, headers);
                resEntity = restTemplate.exchange(url, HttpMethod.GET, httpEntity, String.class);

            }else {
                HttpEntity httpEntity = new HttpEntity<>(headers);
                resEntity = restTemplate.exchange(url, HttpMethod.GET, httpEntity, String.class, param);
            }

            jsonObject = JSONUtil.parseObj(resEntity.getBody());
        } catch (Exception e) {
            if (e.toString().contains("timed out")) {
                throw new BusinessException(ResponseCode.CONNECTION_TIMEOUT);
            }
            log.error("接口调用失败{}",e);
            throw new BusinessException(ResponseCode.GETDATA_FAIL);
        }
        return jsonObject;
    }

    /**
     * 发送请求Get请求
     *
     * @param url   请求接口
     * @param params 请求参数
     */
    public JSONObject sendGet(String url, Map<Object, Object> params){
        StringBuffer stringBuffer = new StringBuffer(url);

        if (params instanceof Map) {
            Iterator iterator = ((Map) params).entrySet().iterator();
            if (iterator.hasNext()) {
                stringBuffer.append("?");
                Object element;
                while (iterator.hasNext()) {
                    element = iterator.next();
                    Map.Entry<String, Object> entry = (Map.Entry) element;
                    if (entry.getValue() != null) {
                        stringBuffer.append(element).append("&");
                    }
                    url = stringBuffer.substring(0, stringBuffer.length() - 1);
                }
            }
        } else {
            throw new BusinessException(ResponseCode.REQUEST_PARAM_ERROR);
        }

        JSONObject jsonObject = null;
        try {
            //设置请求头
            HttpHeaders headers = new HttpHeaders();
            headers.setContentType(MediaType.APPLICATION_JSON);

            ResponseEntity<String> resEntity = null;
            HttpEntity httpEntity = new HttpEntity<>(headers);
            resEntity = restTemplate.exchange(url, HttpMethod.GET, httpEntity, String.class);

            jsonObject = JSONUtil.parseObj(resEntity.getBody());
        } catch (Exception e) {
            if (e.toString().contains("timed out")) {
                throw new BusinessException(ResponseCode.CONNECTION_TIMEOUT);
            }
            log.error("接口调用失败{}",e);
            throw new BusinessException(ResponseCode.GETDATA_FAIL);
        }
        return jsonObject;
    }


    /**
     * 发送请求Put请求
     *
     * @param url   请求接口
     * @param param 请求参数
     */
    public JSONObject sendPut(String url, JSONObject param) {
        JSONObject jsonObject = null;
        try {
            //设置请求头
            HttpHeaders headers = new HttpHeaders();
            headers.setContentType(MediaType.APPLICATION_JSON);

            HttpEntity httpEntity = new HttpEntity<>(param, headers);
            ResponseEntity<String> resEntity = restTemplate.exchange(url, HttpMethod.PUT, httpEntity, String.class);
            jsonObject = JSONUtil.parseObj(resEntity.getBody());
        } catch (Exception e) {
            if (e.toString().contains("timed out")) {
                throw new BusinessException(String.valueOf(ResponseCode.CONNECTION_TIMEOUT));
            }
            log.error("接口调用失败{}",e);
            throw new BusinessException(String.valueOf(ResponseCode.GETDATA_FAIL));
        }
        return jsonObject;
    }

    /**
     * 发送请求Post请求
     *
     * @param url   请求接口
     * @param param 请求参数
     */
    public JSONObject sendPost(String url, Object param) {
        JSONObject jsonObject = null;
        try {
            //设置请求头
            HttpHeaders headers = new HttpHeaders();
            headers.setContentType(MediaType.APPLICATION_JSON);

            HttpEntity httpEntity = new HttpEntity<>(param, headers);
            jsonObject = restTemplate.postForObject(url, httpEntity, JSONObject.class);
        } catch (Exception e) {
            if (e.toString().contains("timed out")) {
                throw new BusinessException(String.valueOf(ResponseCode.CONNECTION_TIMEOUT));
            }
            log.error("接口调用失败{}",e);
            throw new BusinessException(String.valueOf(ResponseCode.GETDATA_FAIL));
        }
        return jsonObject;
    }

    /**
     * 发送请求Post请求
     *
     * @param url   请求接口
     * @param param 请求参数
     */
    public JSONObject sendPost(String url, Object param,Map<String,String> header) {
        JSONObject jsonObject = null;
        try {
            //设置请求头
            HttpHeaders headers = new HttpHeaders();
            headers.setContentType(MediaType.APPLICATION_JSON);
            header.keySet().forEach(item->{
                headers.add(item,header.get(item));
            });

            HttpEntity httpEntity = new HttpEntity<>(param, headers);
            jsonObject = restTemplate.postForObject(url, httpEntity, JSONObject.class);
        } catch (Exception e) {
            if (e.toString().contains("timed out")) {
                throw new BusinessException(String.valueOf(ResponseCode.CONNECTION_TIMEOUT));
            }
            log.error("接口调用失败{}",e);
            throw new BusinessException(String.valueOf(ResponseCode.GETDATA_FAIL));
        }
        return jsonObject;
    }
    
    /**
     * 发送请求Get请求获取流文件
     *
     * @param url   请求接口
     */
    public InputStream sendGet(String url) {
        try {
            ResponseEntity<byte[]> entity = restTemplate.getForEntity(url, byte[].class);
            Objects.requireNonNull(entity.getBody());
            // 下载到临时目录，避免污染 classpath 下的编译输出目录
            File file = File.createTempFile("download-", ".tmp");
            file.deleteOnExit();
            try (BufferedOutputStream bos = new BufferedOutputStream(new FileOutputStream(file))) {
                bos.write(Objects.requireNonNull(entity.getBody()));
            }

            return new FileInputStream(file);
        } catch (Exception e) {
            if (e.toString().contains("timed out")) {
                throw new BusinessException(ResponseCode.CONNECTION_TIMEOUT);
            }
            log.error("接口调用失败{}",e);
            throw new BusinessException(ResponseCode.GETDATA_FAIL);
        }

    }
}

HttpUtil 里有几个值得单独拿出来说的点：

  • 拼接查询参数时用了 StringBuffer，但它在方法内部创建、不存在跨线程共享，用 StringBuilder 更合适；拼接时也没有做 URL 编码，参数里出现 &、空格或中文就会拼出错误的 URL，建议用 UriComponentsBuilder 来构建。
  • 判断超时用的是 e.toString().contains("timed out")，这依赖异常文案，JDK 或 HTTP 客户端升级后文案一变就失效。应当直接 catch (SocketTimeoutException e)，或按 ResourceAccessException 的 cause 类型判断。
  • HttpEntity 没有写泛型，编译只产生警告，但可读性差；建议统一写成 HttpEntity<String>。
  • 方法内 System.out.println 应改为 log.debug；临时文件下载到 classpath 输出目录里，容易污染构建产物，也不便于清理，建议改用 File.createTempFile 并配合 try-with-resources 关流。

JavaScriptUtil

业务里经常需要把一段可配置的规则（比如「公司类型为 1，或者金额大于 100 就返回 true」）做成表达式，而不是每次改需求都发版。JavaScriptUtil 就是让 Java 直接执行一段 JavaScript 片段，并把结果读回来的工具类。

import org.apache.tomcat.util.buf.StringUtils;

import javax.script.*;
import java.util.Map;

/**
 * java执行javaScript代码的工具类
 *
 * @author wuhao
 */
public class JavaScriptUtil {

    /**
     * 解析引擎不是线程安全的，这里用 ThreadLocal 让每个线程持有自己的引擎
     */
    private static final ThreadLocal<ScriptEngine> ENGINE_HOLDER = ThreadLocal.withInitial(() -> {
        ScriptEngine scriptEngine = new ScriptEngineManager().getEngineByName("nashorn");
        if (scriptEngine == null) {
            throw new IllegalStateException("获取JavaScript解析引擎失败");
        }
        return scriptEngine;
    });

    private JavaScriptUtil() {
    }

    private static ScriptEngine newEngine() {
        return ENGINE_HOLDER.get();
    }

    /**
     * 复用线程内引擎时，先清空上一次执行残留的变量，避免脚本之间互相干扰
     */
    private static ScriptEngine freshEngine() {
        ScriptEngine engine = newEngine();
        engine.getBindings(ScriptContext.ENGINE_SCOPE).clear();
        return engine;
    }


    /**
     * 执行一段JavaScript代码
     *
     * @param script JavaScript的代码
     * @return JavaScript代码运行结果的值
     * @throws ScriptException JavaScript代码运行异常
     */
    public static Object execute(String script) throws ScriptException {
        return freshEngine().eval(script);
    }


    /**
     * 运行一个JavaScript代码段,并获取指定变量名的值
     *
     * @param script        代码段
     * @param attributeName 已知的变量名
     * @return 指定变量名对应的值
     * @throws ScriptException JavaScript代码运行异常
     */
    public static Object executeForAttribute(String script, String attributeName) throws ScriptException {
        ScriptEngine engine = freshEngine();
        engine.eval(script);
        return engine.getContext().getAttribute(attributeName);
    }

    /**
     * 获取当前语句运行后第一个有值变量的值
     *
     * @param script 代码段
     * @return 第一个有值变量的值
     * @throws ScriptException JavaScript代码运行异常
     */
    public static Object executeForFirstAttribute(String script) throws ScriptException {

        //每个线程复用一个引擎，执行前清空变量，避免和其他调用的变量互相干扰
        ScriptEngine scriptEngine = freshEngine();

        scriptEngine.eval(script);
        ScriptContext context = scriptEngine.getContext();
        if (context == null) {
            return null;
        }
        Bindings bindings = context.getBindings(ScriptContext.ENGINE_SCOPE);
        if (bindings == null) {
            return null;
        }
        for (Map.Entry<String, Object> entry : bindings.entrySet()) {
            if (entry.getValue() != null) {
                return entry.getValue();
            }
        }
        return null;
    }

    /**
     * 执行一段JavaScript代码，传入参数，获取返回值
     * String script = "Math.pow(x, y)";
     * Map<String, Object> params = Maps.newHashMap();
     * params.put("x", 10);
     * params.put("y", 2);
     *
     * String script = "if (company == 1) { return true; }"
     *         + " else if (company == 2 && amt > 100) { return true; }"
     *         + " else if (company == 3 && amt > 200) { return true; }"
     *         + " else { return false; }";
     *
     * params.put("company", 1);
     * params.put("amt", 100);
     *
     * @param script
     * @param params
     * @return
     */
    public static String execute(String script, Map<String, Object> params) {
        try {
            //注意：参数值由引擎的绑定机制传入，不要直接拼接进脚本文本，否则会有注入风险
            String paramsName = org.apache.commons.lang3.StringUtils.join(params.keySet(), ",");
            String executeScript = String.format("function execute(%s) { %s }", paramsName, script);
            ScriptEngine javaScriptEngine = freshEngine();
            javaScriptEngine.eval(executeScript);
            Invocable jsInvoke = (Invocable) javaScriptEngine;
            Object execute = jsInvoke.invokeFunction("execute", params.values().toArray());
            return String.valueOf(execute);
        } catch (ScriptException | NoSuchMethodException e) {
            //不要把异常吞掉返回 null，否则调用方无法区分「结果为 null」和「执行失败」
            throw new IllegalStateException("表达式执行失败: " + e.getMessage(), e);
        }
    }
}

JavaScriptUtil 有三个必须知道的限制。第一，javax.script.ScriptEngine 不是线程安全的，把它做成 static 单例后，多个请求同时执行脚本会互相覆盖 ScriptContext 里的变量，必须用 ThreadLocal 隔离，或者每次执行前清空绑定。第二，Nashorn 引擎在 JDK 11 被标记废弃、JDK 15 已移除，新项目应当改用 GraalJS 等独立引擎，或直接换成表达式引擎（如 Aviator、QLExpress）。第三，脚本内容来自外部配置时，绝不能用字符串拼接的方式把参数塞进脚本文本，否则等同于打开了代码注入的口子。

MinioUtil

最后一个是对 MinIO 对象存储的封装，承担桶的创建与检查、文件上传、分片合并、预签名 URL 生成这几件事。


import io.minio.*;
import io.minio.errors.*;
import io.minio.http.Method;
import io.minio.messages.Item;
import io.minio.messages.Upload;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;
import org.springframework.util.StringUtils;
import org.springframework.web.multipart.MultipartFile;

import javax.crypto.SecretKey;
import java.io.IOException;
import java.io.InputStream;
import java.security.InvalidKeyException;
import java.security.NoSuchAlgorithmException;
import java.util.ArrayList;
import java.util.List;

@Component
@Slf4j
public class MinioUtil {
    @Autowired
    private MinioClient minioClient;

    private static final int DEFAULT_EXPIRY_TIME = 7 * 24 * 3600;

    /** MinIO 单个对象最多允许合并的分片数 */
    private static final int MAX_COMPOSE_PARTS = 10000;

    /**
     * 检查存储桶是否存在
     *
     * @param bucketName 存储桶名称
     * @return
     * @throws Exception
     */
    public boolean bucketExists(String bucketName)  {
        try {
            return minioClient.bucketExists(BucketExistsArgs.builder().bucket(bucketName).build());
        } catch (Exception e) {
            log.error("mino 服务端连接异常",e);
            throw new RuntimeException("mino 服务端连接异常");
        }
    }

    /**
     * 创建存储桶
     *
     * @param bucketName 存储桶名称
     * @return
     * @throws Exception
     */
    public boolean makeBucket(String bucketName) throws Exception {
        boolean flag = bucketExists(bucketName);
        if (!flag) {
            minioClient.makeBucket(MakeBucketArgs.builder().bucket(bucketName).build());
            return true;
        }
        return false;
    }


    /**
     * 文件上传
     *
     * @param bucketName    存储桶名称
     * @param multipartFile 上传的文件
     * @param filename      对象名
     * @param secretKey     自定义加密密钥，可为 null
     */
    public void putObject(String bucketName, MultipartFile multipartFile, String filename, SecretKey secretKey) throws Exception {
        makeBucket(bucketName);
        PutObjectArgs.Builder builder = PutObjectArgs.builder()
                .bucket(bucketName)
                .object(filename)
                .stream(multipartFile.getInputStream(), multipartFile.getSize(), -1)
                .contentType(multipartFile.getContentType());
        if (null != secretKey) {
            builder.sse(ServerSideEncryption.withCustomerKey(secretKey));
        }
        minioClient.putObject(builder.build());
    }


    /**
     * 普通文件上传
     *
     * @param bucketName  存储桶名称
     * @param inputStream 文件流，由调用方负责关闭
     * @param filename    对象名
     * @param fileSize    文件大小；未知时传 -1，此时必须指定分片大小
     */
    public void putObject(String bucketName, InputStream inputStream, String filename, SecretKey secretKey, long fileSize) throws Exception {
        minioClient.putObject(
                PutObjectArgs.builder()
                        .bucket(bucketName)
                        .object(filename)
                        .stream(inputStream, fileSize, -1)
                        .build());
    }

    /**
     * 分享。
     *
     * @param bucketName 存储桶名称
     * @param objectName 存储桶里的对象名称
     * @param expires    失效时间（以秒为单位），默认是7天，不得大于七天
     * @return 预签名下载地址；桶不存在时返回 null
     */
    public String getPresignedObjectUrl(String bucketName, String objectName, Integer expires) {
        if (!bucketExists(bucketName)) {
            return null;
        }
        if (null == expires) {
            expires = DEFAULT_EXPIRY_TIME;
        }
        if (expires < 1 || expires > DEFAULT_EXPIRY_TIME) {
            throw new IllegalArgumentException("expires must be in range of 1 to " + DEFAULT_EXPIRY_TIME);
        }
        try {
            return minioClient.getPresignedObjectUrl(
                    GetPresignedObjectUrlArgs.builder()
                            .method(Method.GET)
                            .bucket(bucketName)
                            .object(objectName)
                            .expiry(expires)
                            .build());
        } catch (Exception e) {
            log.error("获取预签名地址失败, bucket={}, object={}", bucketName, objectName, e);
            throw new RuntimeException("获取预签名地址失败", e);
        }
    }

    /**
     * 获取一个临时可以上传的url
     *
     * @param bucketName 存储桶名称
     * @param objectName 对象名
     */
    public String presignedPutObject(String bucketName, String objectName) {
        try {
            return minioClient.getPresignedObjectUrl(
                    GetPresignedObjectUrlArgs.builder()
                            .method(Method.PUT)
                            .bucket(bucketName)
                            .object(objectName)
                            .expiry(DEFAULT_EXPIRY_TIME)
                            .build());
        } catch (Exception e) {
            log.error("获取临时文件上传地址异常, bucket={}, object={}", bucketName, objectName, e);
            throw new RuntimeException("获取临时文件上传地址异常", e);
        }
    }

    /**
     * 列出某个存储桶中的所有对象
     *
     * @param bucketName
     * @param objectNamePrefix 前缀
     * @return
     * @throws Exception
     */
    public List<ComposeSource> listObjectsSource(String bucketName, String objectNamePrefix) throws Exception {
        List<ComposeSource> sources = new ArrayList<>();
        Iterable<Result<Item>> results = minioClient.listObjects(
                ListObjectsArgs.builder()
                        .bucket(bucketName)
                        .prefix(objectNamePrefix)
                        .recursive(true)
                        .build());
        for (Result<Item> result : results) {
            Item item = result.get();
            //必须用每个对象真实的 objectName，直接用 prefix 会导致所有分片指向同一个对象
            sources.add(ComposeSource.builder()
                    .bucket(bucketName)
                    .object(item.objectName())
                    .build());
        }
        return sources;
    }

    /**
     * 文件合并
     *
     * 若 sourceObjectList 直接来自 {@link #listObjectsSource}，就是按分片顺序列出后合并。
     * 合并完成后删除分片对象；合并成功后 deleteParts=false 可保留分片以便排查。
     *
     * @param bucketName       存储桶名称
     * @param objectName       合并后的对象名
     * @param sourceObjectList 待合并的分片对象列表
     * @throws IllegalArgumentException 分片数量超出 MinIO 上限（10000）
     */
    public void composeObject(String bucketName, String objectName, List<ComposeSource> sourceObjectList) throws Exception {
        if (sourceObjectList == null || sourceObjectList.size() < 2) {
            throw new IllegalArgumentException("合并至少需要 2 个分片对象");
        }
        if (sourceObjectList.size() > MAX_COMPOSE_PARTS) {
            throw new IllegalArgumentException("分片数量不能超过 " + MAX_COMPOSE_PARTS);
        }
        minioClient.composeObject(
                ComposeObjectArgs.builder()
                        .bucket(bucketName)
                        .object(objectName)
                        .sources(sourceObjectList)
                        .build());
        for (ComposeSource source : sourceObjectList) {
            minioClient.removeObject(
                    RemoveObjectArgs.builder()
                            .bucket(source.bucket())
                            .object(source.object())
                            .build());
        }
    }

    /**
     * 查询分片上传已经成功的分片序号
     *
     * @param bucketName 存储桶名称
     * @param prefix     分片对象名前缀
     * @return 已上传成功的分片序号列表
     */
    public List<Integer> listUploadedParts(String bucketName, String prefix) {
        List<Integer> rslt = new ArrayList<>();
        try {
            Iterable<Result<Item>> results = minioClient.listObjects(
                    ListObjectsArgs.builder()
                            .bucket(bucketName)
                            .prefix(prefix)
                            .maxKeys(1000)
                            .build());
            for (Result<Item> result : results) {
                String objectName = result.get().objectName();
                //按最后一个下划线截取，前缀里本身带下划线时 lenientSplit 也能取对
                String[] split = StringUtils.split(objectName, "_", 2);
                if (null == split) {
                    continue;
                }
                try {
                    rslt.add(Integer.parseInt(split[split.length - 1]));
                } catch (NumberFormatException e) {
                    log.warn("分片对象名不符合命名规范，已跳过: {}", objectName);
                }
            }
        } catch (Exception e) {
            log.error("调用minio查询已经分片上传信息异常", e);
            throw new RuntimeException("调用minio查询已经分片上传信息异常", e);
        }
        return rslt;
    }
}

MinioUtil 最需要注意的是版本一致性。PutObjectOptions、getPresignedObjectUrl(Method, String, String, int, Map)、presignedPutObject 都是 7.x 时代的 API，在 8.x 里已经移除，必须换成 PutObjectArgs、GetPresignedObjectUrlArgs 这类 Args 构造器；而 ListObjectsArgs、ComposeObjectArgs 又是 8.x 才有的写法，两种风格混在一起是编译不过的。此外还有几处容易踩的点：

  • listObjectsSource 里取了 item 却把 objectNamePrefix 当成对象名塞进 ComposeSource，会让所有分片都指向同一个对象，合并结果必然错误；
  • 合并的对象名不能靠 objectName + "_" + i 去猜，分片命名规则应由上传方保证，合并方只负责按 listObjects 返回的顺序合并；
  • listIncompleteUploads 这个名字指的是「列出分片上传过程中产生的对象」，但实现只是按前缀列出对象并解析末尾序号，命名和实现不符，容易误导调用方；
  • 捕获异常后抛 new RuntimeException(msg) 会丢掉 cause，应当传入原始异常，便于定位是网络问题还是权限问题。

三个工具类串起来看，工具类的价值在于把重复的样板逻辑收口，而不是把复杂度藏起来。凡是涉及线程安全（共享引擎）、资源释放（流与临时文件）、版本兼容（第三方 SDK 大版本升级）的地方，都要在工具类里一次性处理干净，否则每个调用方都得重复踩一遍。
